def header(title):

    print(title)

def run_cmd(cmd):
    try:
        out = subprocess.check_output(cmd, stderr=subprocess.STDOUT, text=True)
        return True, out.strip()
    except Exception as e:
        return False, str(e)

def check_import(pkg):
    try:
        m = import_module(pkg)
        ver = getattr(m, "__version__", "unknown")
        return True, ver
    except Exception as e:
        return False, str(e)

def numpy_pandas_smoke_tests():
    import numpy as np
    import pandas as pd

    header("NumPy smoke tests")
    a = np.arange(1, 13).reshape(3, 4)
    print("Array a:\n", a)
    print("Shape:", a.shape)
    print("Mean (axis=0):", a.mean(axis=0))
    mask = a % 2 == 0
    print("Even mask sum:", mask.sum())
    b = a * 2 + 1 # broadcasting-ish arithmetic
    print("b = a*2+1, last row:", b[-1, :])
    
    header("Pandas smoke tests")
    df = pd.DataFrame({
        "id": [1, 2, 3, 4, 5],
        "group": ["A", "A", "B", "B", "B"],
        "value": [10, 12, 9, None, 15]
    })
    print("DF:\n", df)
    print("\nMissing values per column:\n", df.isna().sum())
    
    # basic cleaning + groupby summary
    df["value_filled"] = df["value"].fillna(df["value"].median())
    summary = df.groupby("group")["value_filled"].agg(["count", "mean", "min", "max"])
    print("\nGroup summary:\n", summary)

    # save + reload to ensure IO works
    out_csv = "lab2_test_output.csv"
    df.to_csv(out_csv, index=False)
    df2 = pd.read_csv(out_csv)
    print("\nReloaded CSV head:\n", df2.head())

def main():
    header("System info")
    print("Python:", sys.version.replace("\n", " "))
    print("Executable:", sys.executable)
    print("Platform:", platform.platform())

    header("Package import checks")
    ok_all = True
    for pkg in REQUIRED:
        ok, info = check_import(pkg)
        print(f"{pkg:10s}:", "OK" if ok else "FAIL", "|", info)
        ok_all = ok_all and ok

    header("Jupyter checks")
    ok_jup, jup_ver = run_cmd(["jupyter", "--version"])
    print("jupyter --version:", "OK" if ok_jup else "FAIL")
    print(jup_ver)

    ok_lab, lab_ver = run_cmd(["jupyter", "lab", "--version"])
    print("\njupyter lab --version:", "OK" if ok_lab else "FAIL")
    print(lab_ver)

    # Kernel list (nice-to-have; some systems may block)
    ok_k, kernels = run_cmd(["jupyter", "kernelspec", "list"])
    print("\njupyter kernelspec list:", "OK" if ok_k else "FAIL")

if __name__ == "__main__":

📊 Good vs Bad Viz (Yunsik)

Bad visualization

(Source: Geospatial Science Lab, Dongguk University)


Domain

  • This example shows urban sprawl and green space changes in Sejong City, South Korea, by comparing 2019 with 2001
  • The implied goal is to understand how urban expansion has affected forest and green spaces over time


Data and task abstraction

  • Raster-based land cover data are used as the primary data source
  • The upper maps appear to show all land cover classes, while the lower maps focus on green space categories
  • A table summarizes proportions of each green space type and differences between 2001 and 2019
  • Green space change is quantified only in the table, whereas urban sprawl is shown only visually
  • Because there is no explicit change map, viewers must mentally compare the 2001 and 2019 maps to infer changes


Encoding and interaction

  • The maps suffer from resolution and clarity issues, making it difficult to interpret fine-scale spatial patterns
  • It is unclear whether the table summarizing green space change refers to the entire study area or only the inset area
  • The purpose of the inset maps is not explained (e.g., why this areas is important, or what role it plays in the urban system)


Algorithm design

  • Land cover classification procedure was conducted on raster data
  • The proportion of each land cover class was calculated for both years
  • However, these computations are not effectively translated into interpretable visual outputs


Suggestion

  • Adjusting the relative scale and proportion between the overview maps and the inset maps
  • Including short descriptive annotations explaining (1) why the highlighted area matters, and (2) what the observed changes imply in an urban or ecological context
  • The layout contains excessive unused space, indicating a need for more efficient spatial arrangement of map elements

Good visualization

(Source: Geospatial Science Lab, Dongguk University)


Domain

  • This example shows ranking of quality of life for the elderly in Seoul, South Korea
  • Although the research question is not explicitly stated, it can be more clearly inferred, compared to the bad example, such as: (1) How does the quality of life of elderly populations vary across districts in Seoul? (2) Which district perform relatively better or worse?
  • The analytical purpose is immediately understandable to the user.


Data and task abstraction

  • Quality of life is compared at the district (gu) level, and multiple indicators are synthesized into a single composite index
  • Instead of listing all 25 districts, the summary focuses on the Top 5 and Bottom 5 districts
  • Data are aggregated as areal (polygon) units, and continuous variables are summarized using a composite z-score
  • This abstraction supports comparison, ranking, and synthesis


Encoding and interaction

  • Areas with high and low quality of life are distinguishable through the use of a choropleth map
  • Visual hierarchy is established: (1) The composite z-score map is the biggest element; (2) Supporting maps of variables are shown at a smaller scale
  • The Top / Bottom 5 ranking lists improve legibility without overwhelming the viewer
  • The color scheme and number of classes are appropriate for representing continuous data


Algorithm design

  • The z-score standardization process is explicitly documented using equations, which supports reproducibility
  • Map projection, data sources, and other technical details are clearly specified


Suggestion

  • The reference year of the data is not explicitly stated and should be included for temporal clarity.
  • The river and stream layer could be retained only in the final composite map, as it is not essential in the supporting maps.

# Write GeoTIFFs with rasterio (keeps CRS, transform, etc.)
os.makedirs('outputs', exist_ok=True)

def write_tif(path, arr, profile):
    prof = profile.copy()
    prof.update(dtype='float32', count=1, nodata=np.nan)
    with rio.open(path, 'w', **prof) as dst:
        dst.write(arr.astype('float'), 1)

write_tif('outputs/slope_degrees.tif', np.array(slope_deg), profile)
write_tif('outputs/aspect_degrees.tif', np.array(aspect), profile)
write_tif('outputs/hillshade.tif', hill.astype('float32'), profile)

print('Wrote GeoTIFFs to outputs/')
